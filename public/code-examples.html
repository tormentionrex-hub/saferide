import java.net.http.*;
import java.net.URI;
import com.google.gson.*;

public class SafeRideAPI {
    private static final String BASE_URL = "https://app-saferide-api-dev.azurewebsites.net/api/v1";
    
    public static void registerUser() throws Exception {
        HttpClient client = HttpClient.newHttpClient();
        
        // Crear JSON payload
        JsonObject payload = new JsonObject();
        payload.addProperty("name", "Juan Pérez");
        payload.addProperty("username", "juanperez");
        payload.addProperty("email", "juan.perez@company.com");
        payload.addProperty("phone", "88001234");
        payload.addProperty("password", "MiPassword@123");
        payload.addProperty("company", "Intel Costa Rica");
        payload.addProperty("type", "pasajero");
        
        // Crear request
        HttpRequest request = HttpRequest.newBuilder()
            .uri(URI.create(BASE_URL + "/auth/register"))
            .header("Content-Type", "application/json")
            .POST(HttpRequest.BodyPublishers.ofString(payload.toString()))
            .build();
        
        // Enviar request
        HttpResponse response = client.send(request, 
            HttpResponse.BodyHandlers.ofString());
        
        // Procesar respuesta
        Gson gson = new Gson();
        JsonObject data = gson.fromJson(response.body(), JsonObject.class);
        
        if (data.get("success").getAsBoolean()) {
            String accessToken = data.getAsJsonObject("data")
                .get("accessToken").getAsString();
            System.out.println("Usuario registrado exitosamente");
            System.out.println("Access Token: " + accessToken);
        }
    }
}

const searchRides = async (origin, destination, date) => {
  const params = new URLSearchParams({
    origin: `${origin.lat},${origin.lng}`,
    destination: `${destination.lat},${destination.lng}`,
    date: date,
    minSeats: 2,
    page: 1,
    limit: 10
  });
  
  const response = await fetch(
    `https://app-saferide-api-dev.azurewebsites.net/api/v1/rides?${params}`
  );
  
  const data = await response.json();
  
  if (data.success) {
    console.log(`Encontrados ${data.pagination.total} viajes`);

import requests
from datetime import datetime

def search_rides(origin, destination, date, access_token=None):
    url = 'https://app-saferide-api-dev.azurewebsites.net/api/v1/rides'
    
    params = {
        'origin': f"{origin['lat']},{origin['lng']}",
        'destination': f"{destination['lat']},{destination['lng']}",
        'date': date,
        'minSeats': 2,
        'page': 1,
        'limit': 10
    }
    
    headers = {}
    if access_token:
        headers['Authorization'] = f'Bearer {access_token}'
    
    response = requests.get(url, params=params, headers=headers)
    data = response.json()
    
    if data['success']:
        print(f"Encontrados {data['pagination']['total']} viajes")
        for ride in data['data']:
            print(f"""
                Viaje: {ride['origin']['name']} → {ride['destination']['name']}
                Conductor: {ride['driver_id']['name']}
                Fecha: {ride['date_time']}
                Asientos: {ride['available_seats']}
                Distancia: {ride['distance_km']} km
            """)
        return data['data']
    return []

const createBooking = async (rideId, seatsBooked = 1) => {
  const accessToken = localStorage.getItem('accessToken');

    {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${accessToken}`
      },
      body: JSON.stringify({
        ride_id: rideId,
        seats_booked: seatsBooked
      })
    }
  );
  
  const data = await response.json();
  
  if (data.success) {
    console.log('✅ Reservación creada exitosamente');
    console.log('ID de reservación:', data.data._id);
    console.log('Estado:', data.data.status);
  } else {
    console.error('❌ Error:', data.message);
  }
};

def create_booking(ride_id, access_token, seats_booked=1):
    url = 'https://app-saferide-api-dev.azurewebsites.net/api/v1/bookings'
    
    headers = {
        'Content-Type': 'application/json',
        'Authorization': f'Bearer {access_token}'
    }
    
    payload = {
        'ride_id': ride_id,
        'seats_booked': seats_booked
    }
    
    response = requests.post(url, json=payload, headers=headers)
    data = response.json()
    
    if data['success']:
        print('✅ Reservación creada exitosamente')
        print(f"ID: {data['data']['_id']}")
        print(f"Estado: {data['data']['status']}")
        return data['data']
    else:
        print(f'❌ Error: {data["message"]}')
        return None
